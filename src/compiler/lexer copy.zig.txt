const std = @import("std");

const token = @import("token.zig");
const lib = @import("lib");

const Source = lib.Source;

const TokenKind = token.TokenKind;
const Token = token.Token;

pub const Part = struct {
    found: []const u8,
    start: u32,
    len: u32,

    pub fn slice(self: Part, source: []const u8) []const u8 {
        return source[self.start .. self.start + self.len];
    }
};

// ── Keyword map ───────────────────────────────────────────────────────────────

const keywords = std.StaticStringMap(TokenKind).initComptime(.{
    .{ "const", .const_kw },
    .{ "val", .val_kw },
    .{ "set", .set_kw },
    .{ "arg", .arg_kw },
    .{ "task", .task_kw },
    .{ "group", .group_kw },
    .{ "if", .if_kw },
    .{ "else", .else_kw },
    .{ "and", .and_kw },
    .{ "or", .or_kw },
    .{ "not", .not_kw },
    .{ "true", .true_kw },
    .{ "false", .false_kw },
    .{ "string", .string_type },
    .{ "flag", .flag_type },
    .{ "int", .int_type },
    .{ "number", .number_type },
    .{ "list", .list_type },
});

// ── Lexer ─────────────────────────────────────────────────────────────────────

pub const Lexer = struct {
    source: *Source,

    pub fn init(source: *Source) Lexer {
        return .{ .source = source };
    }

    // ── Public interface ──────────────────────────────────────────────────────

    pub fn next(self: *Lexer) Token {
        while (true) {
            self.skipWhitespace();

            const ch = self.source.peek();

            switch (ch) {
                // Comments — skip and loop
                0 => {
                    return self.makeToken(.eof, self.pos, 0);
                },
                '/' => {
                    if (self.peekAt(1) == '/') {
                        self.skipLineComment();
                        continue;
                    }
                    // Single / is always invalid in TSQ
                    return self.consumeInvalid();
                },

                // Single-char symbols
                '{' => return self.consumeChar(.lbrace),
                '}' => return self.consumeChar(.rbrace),
                '[' => return self.consumeChar(.lbracket),
                ']' => return self.consumeChar(.rbracket),
                '(' => return self.consumeChar(.lparen),
                ')' => return self.consumeChar(.rparen),
                ':' => return self.consumeChar(.colon),
                '.' => return self.consumeChar(.dot),
                '+' => return self.consumeChar(.plus),
                '-' => return self.consumeChar(.minus),
                '*' => return self.consumeChar(.star),

                // Two-char symbols
                '=' => {
                    if (self.source.peekAt(1) == '=') return self.consumeTwo(.eq_eq);
                    return self.consumeChar(.eq);
                },
                '!' => {
                    if (self.source.peekAt(1) == '=') return self.consumeTwo(.bang_eq);
                    return self.consumeInvalid();
                },
                '<' => {
                    if (self.source.peekAt(1) == '=') return self.consumeTwo(.lt_eq);
                    return self.consumeChar(.lt);
                },
                '>' => {
                    if (self.source.peekAt(1) == '=') return self.consumeTwo(.gt_eq);
                    return self.consumeChar(.gt);
                },

                // String literal
                '"' => return self.lexString(),

                // Backtick process line
                '`' => return self.lexBacktick(),

                // Numbers
                '0'...'9' => return self.lexNumber(),

                // Identifiers and keywords
                'a'...'z', 'A'...'Z', '_' => return self.lexIdent(),

                // Everything else
                else => return self.consumeInvalid(),
            }
        }
    }

    //TODO: handle escape character
    pub fn readTo(self: *Lexer, limiters: [][]const u8) Part {
        self.skipWhitespace();

        const start = self.pos;

        while (self.pos < self.source.len) : (self.pos += 1) {
            center: for (limiters) |limiter| {
                for (limiter, 0..) |ch, idx| {
                    if (self.source.peekAt(@intCast(idx)) != ch)
                        continue :center;
                }

                return .{
                    .found = limiter,
                    .start = start,
                    .len = self.pos - start + limiter.len,
                };
            }
        }

        return .{
            .found = .{},
            .start = self.pos,
            .len = 0,
        };
    }

    // ── Private helpers ───────────────────────────────────────────────────────

    // fn peek(self: *Lexer) u8 {
    //     if (self.pos >= self.source.len) return 0;
    //     return self.source[self.pos];
    // }

    // fn peekAt(self: *Lexer, offset: u32) u8 {
    //     const i = self.pos + offset;
    //     if (i >= self.source.len) return 0;
    //     return self.source[i];
    // }

    fn skipWhitespace(self: *Lexer) void {
        while (!self.source.eof()) {
            switch (self.source.read()) {
                ' ', '\t', '\r', '\n' => self.pos += 1,
                else => break,
            }
        }
    }

    fn skipLineComment(self: *Lexer) void {
        // caller already confirmed source[pos..pos+2] == "//"
        while (self.pos < self.source.len and self.source[self.pos] != '\n') {
            self.pos += 1;
        }
    }

    // ── Token constructors ────────────────────────────────────────────────────

    fn makeToken(self: *Lexer, kind: TokenKind, start: u32, len: u32) Token {
        _ = self;
        return .{ .kind = kind, .start = start, .len = len };
    }

    fn consumeChar(self: *Lexer, kind: TokenKind) Token {
        const start = self.source.pos;
        self.source.pos += 1;
        return self.makeToken(kind, start, 1);
    }

    fn consumeTwo(self: *Lexer, kind: TokenKind) Token {
        const start = self.source.pos;
        self.source.pos += 2;
        return self.makeToken(kind, start, 2);
    }

    fn consumeInvalid(self: *Lexer) Token {
        const start = self.source.pos;
        self.source.pos += 1;
        return self.makeToken(.invalid_char, start, 1);
    }

    // ── Scanners ──────────────────────────────────────────────────────────────

    fn lexIdent(self: *Lexer) Token {
        const start = self.source.pos;
        while (self.source.pos < self.source.len) {
            switch (self.source.peek()) {
                'a'...'z', 'A'...'Z', '0'...'9', '_' => self.source.pos += 1,
                else => break,
            }
        }
        const text = self.source.substring(
            start,
            self.source.pos,
        );
        const kind = keywords.get(text) orelse .ident;
        return self.makeToken(kind, start, self.source.pos - start);
    }

    fn lexNumber(self: *Lexer) Token {
        const start = self.source.pos;
        var is_float = false;

        while (self.source.pos < self.source.len and isDigit(self.source.peek())) {
            self.source.pos += 1;
        }

        // Check for decimal point followed by more digits
        if (self.source.pos < self.source.len and
            self.source.peek() == '.' and
            self.source.pos + 1 < self.source.len and
            isDigit(self.source.peekAt(1)))
        {
            is_float = true;
            self.source.pos += 1; // consume .
            while (self.source.pos < self.source.len and isDigit(self.source.peek())) {
                self.source.pos += 1;
            }
        }

        const kind: TokenKind = if (is_float) .float else .int;
        return self.makeToken(kind, start, self.pos - start);
    }

    fn lexString(self: *Lexer) Token {
        const start = self.source.pos;
        self.source.pos += 1; // consume opening "

        while (self.source.pos < self.source.len) {
            switch (self.source.peek()) {
                '"' => {
                    self.source.pos += 1; // consume closing "
                    return self.makeToken(.string, start, self.source.pos - start);
                },
                '\\' => {
                    // skip backslash + next char — escape validation is
                    // a semantic concern, not a lexer concern
                    self.source.pos += 1;
                    if (self.source.pos < self.source.len) self.source.pos += 1;
                },
                '\n' => {
                    // do NOT consume the newline — let next call start cleanly
                    return self.makeToken(.unterminated_string, start, self.source.pos - start);
                },
                else => self.source.pos += 1,
            }
        }

        // reached EOF without closing quote
        return self.makeToken(.unterminated_string, start, self.source.pos - start);
    }

    fn lexBacktick(self: *Lexer) Token {
        const start = self.source.pos;
        self.source.pos += 1; // consume opening `

        while (self.source.pos < self.source.len) {
            switch (self.source.peek()) {
                '`' => {
                    self.source.pos += 1; // consume closing `
                    return self.makeToken(.backtick, start, self.source.pos - start);
                },
                '\n' => {
                    // backtick lines must be single-line
                    return self.makeToken(.unterminated_backtick, start, self.source.pos - start);
                },
                '\\' => {
                    // allow escaping backtick itself: \`
                    self.source.pos += 1;
                    if (self.source.pos < self.source.len) self.source.pos += 1;
                },
                else => self.source.pos += 1,
            }
        }

        return self.makeToken(.unterminated_backtick, start, self.source.pos - start);
    }

    // ── Char utils ────────────────────────────────────────────────────────────

    fn isDigit(ch: u8) bool {
        return ch >= '0' and ch <= '9';
    }
};

// ── Tests ─────────────────────────────────────────────────────────────────────

test "eof on empty input" {
    var source = Source.init("");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.eof, lex.next().kind);
}

test "single char tokens" {
    var source = Source.init("{}[]():.");
    var lex = Lexer.init(&source);
    const expected = [_]TokenKind{
        .lbrace, .rbrace, .lbracket, .rbracket,
        .lparen, .rparen, .colon,    .dot,
    };
    for (expected) |kind| {
        try std.testing.expectEqual(kind, lex.next().kind);
    }
    try std.testing.expectEqual(TokenKind.eof, lex.next().kind);
}

test "two char tokens" {
    var source = Source.init("== != <= >=");
    var lex = Lexer.init(&source);
    const expected = [_]TokenKind{ .eq_eq, .bang_eq, .lt_eq, .gt_eq };
    for (expected) |kind| {
        try std.testing.expectEqual(kind, lex.next().kind);
    }
}

test "single eq is not eq_eq" {
    var source = Source.init("= ==");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.eq, lex.next().kind);
    try std.testing.expectEqual(TokenKind.eq_eq, lex.next().kind);
}

test "keywords" {
    var source = Source.init("const val set arg task group if else and or not true false");
    var lex = Lexer.init(&source);
    const expected = [_]TokenKind{
        .const_kw, .val_kw,   .set_kw, .arg_kw,
        .task_kw,  .group_kw, .if_kw,  .else_kw,
        .and_kw,   .or_kw,    .not_kw, .true_kw,
        .false_kw,
    };
    for (expected) |kind| {
        try std.testing.expectEqual(kind, lex.next().kind);
    }
}

test "type keywords" {
    var source = Source.init("string flag int number list");
    var lex = Lexer.init(&source);
    const expected = [_]TokenKind{
        .string_type, .flag_type, .int_type, .number_type, .list_type,
    };
    for (expected) |kind| {
        try std.testing.expectEqual(kind, lex.next().kind);
    }
}

test "keyword prefix is not a keyword" {
    // 'constant' starts with 'const' but is an ident
    var source = Source.init("constant tasker grouped");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
}

test "ident slice" {
    var source = Source.init("hello_world");
    var lex = Lexer.init(&source);
    const tok = lex.next();
    try std.testing.expectEqual(TokenKind.ident, tok.kind);
    try std.testing.expectEqualStrings("hello_world", tok.slice(source.buffer));
}

test "integer" {
    var source = Source.init("42");
    var lex = Lexer.init(&source);
    const tok = lex.next();
    try std.testing.expectEqual(TokenKind.int, tok.kind);
    try std.testing.expectEqualStrings("42", tok.slice(source.buffer));
}

test "float" {
    var source = Source.init("3.14");
    var lex = Lexer.init(&source);
    const tok = lex.next();
    try std.testing.expectEqual(TokenKind.float, tok.kind);
    try std.testing.expectEqualStrings("3.14", tok.slice(source.buffer));
}

test "dot between ints is not float" {
    // 42.foo should lex as int, dot, ident — not float
    var source = Source.init("42.foo");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.int, lex.next().kind);
    try std.testing.expectEqual(TokenKind.dot, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
}

test "string" {
    var source = Source.init("\"hello world\"");
    var lex = Lexer.init(&source);
    const tok = lex.next();
    try std.testing.expectEqual(TokenKind.string, tok.kind);
    try std.testing.expectEqualStrings("\"hello world\"", tok.slice(source.buffer));
}

test "string with escape" {
    var source = Source.init("\"say \\\"hi\\\"\"");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.string, lex.next().kind);
}

test "unterminated string at newline" {
    var source = Source.init("\"hello\nworld");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.unterminated_string, lex.next().kind);
    // lexer should recover — next token is the ident on the new line
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
}

test "unterminated string at eof" {
    var source = Source.init("\"hello");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.unterminated_string, lex.next().kind);
    try std.testing.expectEqual(TokenKind.eof, lex.next().kind);
}

test "backtick" {
    var source = Source.init("`echo {{name}}`");
    var lex = Lexer.init(&source);
    const tok = lex.next();
    try std.testing.expectEqual(TokenKind.backtick, tok.kind);
    try std.testing.expectEqualStrings("`echo {{name}}`", tok.slice(source.buffer));
}

test "unterminated backtick" {
    var source = Source.init("`echo hello\nnext line");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.unterminated_backtick, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind); // "next"
}

test "line comment is skipped" {
    var source = Source.init("// this is a comment\nconst");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.const_kw, lex.next().kind);
}

test "single slash is invalid" {
    var source = Source.init("/ const");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.invalid_char, lex.next().kind);
    try std.testing.expectEqual(TokenKind.const_kw, lex.next().kind);
}

test "invalid char recovers" {
    var source = Source.init("@ const");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.invalid_char, lex.next().kind);
    try std.testing.expectEqual(TokenKind.const_kw, lex.next().kind);
}

test "attribute tokens" {
    // [shell] lexes as lbracket ident rbracket
    var source = Source.init("[shell]");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.lbracket, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
    try std.testing.expectEqual(TokenKind.rbracket, lex.next().kind);
}

test "attribute with value" {
    // [short = "x"] lexes as lbracket ident eq string rbracket
    var source = Source.init("[short = \"x\"]");
    var lex = Lexer.init(&source);
    try std.testing.expectEqual(TokenKind.lbracket, lex.next().kind);
    try std.testing.expectEqual(TokenKind.ident, lex.next().kind);
    try std.testing.expectEqual(TokenKind.eq, lex.next().kind);
    try std.testing.expectEqual(TokenKind.string, lex.next().kind);
    try std.testing.expectEqual(TokenKind.rbracket, lex.next().kind);
}

test "full snippet" {
    const expected = [_]TokenKind{
        .group_kw,  .ident,    .lbrace,
        .arg_kw,    .ident,    .colon,
        .flag_type, .eq,       .false_kw,
        .const_kw,  .ident,    .eq,
        .string,    .backtick, .rbrace,
        .eof,
    };

    var source = Source.init(
        \\group go {
        \\    arg dev : flag = false
        \\    const path = "./bin/"
        \\    `go build -o {{path}}`
        \\}
    );

    var lex = Lexer.init(&source);
    for (expected) |kind| {
        const tok = lex.next();
        try std.testing.expectEqual(kind, tok.kind);
    }
}
